% 8. Mô hình chính
\sfoot{$P(i,j)$: các ứng viên gần $i$ hơn hoặc bằng $j$ · $q_i$: nhu cầu điểm $i$ ·
  $Q_j$: công suất trạm $j$}
\begin{frame}{MILP với hai biến nhị phân và năm ràng buộc}
  \framesubtitle{MÔ HÌNH CHÍNH}
  {\Flead\color{soft}$x_j = 1$ nếu mở trạm $j$.\; $y_{ij} = 1$ nếu điểm cầu $i$ được gán cho
    trạm $j$, chỉ xét các cặp có khoảng cách không quá R.\par}
  \vspace{3mm}
  \begin{card}[left=4mm, right=4mm, top=0.6mm, bottom=0.6mm]
    \fontsize{8.5}{10.5}\selectfont\renewcommand{\arraystretch}{2.05}\arrayrulecolor{hair}
    \begin{tabular}{@{}L{56mm}@{\hspace{4mm}}L{72mm}@{}}
      \textcolor{ink}{$\sum_{j \in N_i} y_{ij} \le 1$} & Mỗi điểm cầu gán tối đa một trạm \\ \hline
      \textcolor{ink}{$y_{ij} \le x_j$} & Chỉ gán vào trạm đã mở \\ \hline
      \textcolor{accent}{$x_j \le \sum_{k \in P(i,j)} y_{ik}$} & \textbf{Gán về trạm mở gần nhất} \\ \hline
      \textcolor{ink}{$\sum_i q_i\, y_{ij} \le Q_j\, x_j$} & Không vượt công suất ngày của trạm \\ \hline
      \textcolor{ink}{$\sum_{i,j} q_i\, y_{ij} \ge \alpha \sum_i q_i$} & Phủ ít nhất α tổng nhu cầu \\
    \end{tabular}
  \end{card}
  \note{Mô hình chính là bài toán quy hoạch nguyên hỗn hợp. Biến x cho biết có mở trạm hay không,
    biến y cho biết điểm dân nào về trạm nào. Có năm ràng buộc, trong đó ràng buộc thứ ba là điểm
    khác biệt chính so với ba mô hình đối chứng: người dân phải đi về trạm mở gần nhất, mô hình
    không được tự ý điều họ sang trạm xa hơn. Giải bằng scipy với solver HiGHS.}
\end{frame}
